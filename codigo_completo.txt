package com.hfad.started_laboratorio;

import android.app.IntentService;
import android.content.Intent;
import android.util.Log;

/**
 * Parte 1 de la guÃ­a: started service que espera 10 s y luego muestra
 * el texto en el logcat y en una notificaciÃ³n.
 * (IntentService estÃ¡ deprecado desde API 30, pero es lo que pide el laboratorio.)
 */
@SuppressWarnings("deprecation")
public class DelayedMessageService extends IntentService {

    public static final String EXTRA_MESSAGE = "message";
    public static final int NOTIFICATION_ID = 5453;
    private static final String TAG = "DelayedMessageService";

    public DelayedMessageService() {
        super("DelayedMessageService");
    }

    @Override
    protected void onHandleIntent(Intent intent) {
        try {
            Thread.sleep(10000); // 10 segundos, en un hilo aparte
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            return;
        }
        String text = intent.getStringExtra(EXTRA_MESSAGE);
        showText(text);
    }

    private void showText(final String text) {
        Log.v(TAG, "The message is: " + text);
        NotificationHelper.show(this, NOTIFICATION_ID,
                getString(R.string.question), text);
    }
}

package com.hfad.started_laboratorio;

import android.app.IntentService;
import android.content.Intent;
import android.util.Log;

/**
 * Parte 2 (funcionalidad adicional): notificaciones del contexto educativo.
 *  - TYPE_GRADE:      el estudiante recibe la calificaciÃ³n de un taller o parcial.
 *  - TYPE_SUBMISSION: el profesor recibe aviso de que una tarea fue entregada
 *                     y ya puede calificarla.
 */
@SuppressWarnings("deprecation")
public class EducationNotificationService extends IntentService {

    public static final String EXTRA_TYPE = "type";
    public static final String EXTRA_ACTIVITY_NAME = "activity_name";
    public static final String EXTRA_VALUE = "value"; // nota o nombre del estudiante

    public static final String TYPE_GRADE = "GRADE";
    public static final String TYPE_SUBMISSION = "SUBMISSION";

    private static final String TAG = "EducationNotification";

    public EducationNotificationService() {
        super("EducationNotificationService");
    }

    @Override
    protected void onHandleIntent(Intent intent) {
        try {
            Thread.sleep(5000); // simula el proceso (calificar / subir tarea)
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            return;
        }

        String type = intent.getStringExtra(EXTRA_TYPE);
        String activity = intent.getStringExtra(EXTRA_ACTIVITY_NAME);
        String value = intent.getStringExtra(EXTRA_VALUE);

        String title;
        String text;
        if (TYPE_GRADE.equals(type)) {
            title = getString(R.string.notif_grade_title);
            text = getString(R.string.notif_grade_text, activity, value);
        } else {
            title = getString(R.string.notif_submission_title);
            text = getString(R.string.notif_submission_text, value, activity);
        }

        Log.v(TAG, title + " - " + text);
        // id distinto por notificaciÃ³n para que no se reemplacen entre sÃ­
        NotificationHelper.show(this, (int) System.currentTimeMillis(), title, text);
    }
}

package com.hfad.started_laboratorio;

import android.Manifest;
import android.content.Intent;
import android.content.pm.PackageManager;
import android.os.Build;
import android.os.Bundle;
import android.view.View;
import android.widget.EditText;
import android.widget.Toast;

import androidx.activity.result.ActivityResultLauncher;
import androidx.activity.result.contract.ActivityResultContracts;
import androidx.appcompat.app.AppCompatActivity;
import androidx.core.content.ContextCompat;

public class MainActivity extends AppCompatActivity {

    private EditText etActivityName;
    private EditText etValue;

    // Android 13+ (API 33) exige pedir permiso para mostrar notificaciones
    private final ActivityResultLauncher<String> permissionLauncher =
            registerForActivityResult(new ActivityResultContracts.RequestPermission(),
                    granted -> {
                        if (!granted) {
                            Toast.makeText(this, R.string.permission_denied,
                                    Toast.LENGTH_LONG).show();
                        }
                    });

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        etActivityName = findViewById(R.id.etActivityName);
        etValue = findViewById(R.id.etValue);

        NotificationHelper.createChannel(this);
        requestNotificationPermission();
    }

    private void requestNotificationPermission() {
        if (Build.VERSION.SDK_INT >= 33
                && ContextCompat.checkSelfPermission(this,
                Manifest.permission.POST_NOTIFICATIONS) != PackageManager.PERMISSION_GRANTED) {
            permissionLauncher.launch(Manifest.permission.POST_NOTIFICATIONS);
        }
    }

    /** BotÃ³n de la guÃ­a: inicia DelayedMessageService con el texto "Timing!". */
    public void onClick(View view) {
        Intent intent = new Intent(this, DelayedMessageService.class);
        intent.putExtra(DelayedMessageService.EXTRA_MESSAGE, getString(R.string.button_response));
        startService(intent);
    }

    /** Estudiante: llega la calificaciÃ³n de un taller o parcial. */
    public void onGradeClick(View view) {
        startEducationService(EducationNotificationService.TYPE_GRADE);
    }

    /** Profesor: un estudiante entregÃ³ una tarea y hay que calificarla. */
    public void onSubmissionClick(View view) {
        startEducationService(EducationNotificationService.TYPE_SUBMISSION);
    }

    private void startEducationService(String type) {
        String activityName = etActivityName.getText().toString().trim();
        String value = etValue.getText().toString().trim();
        if (activityName.isEmpty() || value.isEmpty()) {
            Toast.makeText(this, R.string.fill_fields, Toast.LENGTH_SHORT).show();
            return;
        }
        Intent intent = new Intent(this, EducationNotificationService.class);
        intent.putExtra(EducationNotificationService.EXTRA_TYPE, type);
        intent.putExtra(EducationNotificationService.EXTRA_ACTIVITY_NAME, activityName);
        intent.putExtra(EducationNotificationService.EXTRA_VALUE, value);
        startService(intent);
        Toast.makeText(this, R.string.notification_scheduled, Toast.LENGTH_SHORT).show();
    }
}

package com.hfad.started_laboratorio;

import android.app.NotificationChannel;
import android.app.NotificationManager;
import android.app.PendingIntent;
import android.content.Context;
import android.content.Intent;
import android.os.Build;

import androidx.core.app.NotificationCompat;

/** Centraliza la creaciÃ³n del canal y la emisiÃ³n de notificaciones. */
public final class NotificationHelper {

    public static final String CHANNEL_ID = "joke_channel";

    private NotificationHelper() {}

    /** Desde Android 8 (API 26) toda notificaciÃ³n necesita un canal. */
    public static void createChannel(Context context) {
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            NotificationChannel channel = new NotificationChannel(
                    CHANNEL_ID,
                    context.getString(R.string.channel_name),
                    NotificationManager.IMPORTANCE_HIGH);
            NotificationManager nm = context.getSystemService(NotificationManager.class);
            nm.createNotificationChannel(channel);
        }
    }

    public static void show(Context context, int notificationId, String title, String text) {
        createChannel(context);

        // PendingIntent: al tocar la notificaciÃ³n se abre MainActivity
        Intent actionIntent = new Intent(context, MainActivity.class);
        PendingIntent pendingIntent = PendingIntent.getActivity(
                context, 0, actionIntent,
                PendingIntent.FLAG_UPDATE_CURRENT | PendingIntent.FLAG_IMMUTABLE);

        NotificationCompat.Builder builder =
                new NotificationCompat.Builder(context, CHANNEL_ID)
                        .setSmallIcon(android.R.drawable.ic_dialog_info)
                        .setContentTitle(title)
                        .setContentText(text)
                        .setStyle(new NotificationCompat.BigTextStyle().bigText(text))
                        .setPriority(NotificationCompat.PRIORITY_HIGH)
                        .setVibrate(new long[]{0, 1000})
                        .setAutoCancel(true)
                        .setContentIntent(pendingIntent);

        NotificationManager nm =
                (NotificationManager) context.getSystemService(Context.NOTIFICATION_SERVICE);
        nm.notify(notificationId, builder.build());
    }
}
